// Database aturan panjang nomor rekening bank di Indonesia const bankRules = { bca: { name: "BCA", minLength: 10, maxLength: 10 }, mandiri: { name: "Mandiri", minLength: 13, maxLength: 13 }, bri: { name: "BRI", minLength: 15, maxLength: 15 }, bni: { name: "BNI", minLength: 10, maxLength: 10 }, cimb: { name: "CIMB Niaga", minLength: 10, maxLength: 13 }, danamon: { name: "Danamon", minLength: 10, maxLength: 12 }, permata: { name: "Permata", minLength: 10, maxLength: 10 }, bsi: { name: "BSI (Bank Syariah Indonesia)", minLength: 10, maxLength: 15 }, // Tambahkan bank lain sesuai kebutuhan }; /** * Fungsi untuk memvalidasi nomor rekening bank * @param {string} bankCode - Kode bank (contoh: 'bca', 'mandiri', 'bri') * @param {string} accountNumber - Nomor rekening yang akan di significativa/validasi * @returns {object} - Hasil validasi (status boolean dan pesan) */ function validateBankAccount(bankCode, accountNumber) { // 1. Bersihkan spasi dan pastikan hanya berisi angka const cleanAccountNo = accountNumber.trim(); const isNumeric = /^\d+$/.test(cleanAccountNo); if (!isNumeric) { return { isValid: false, message: "Nomor rekening hanya boleh berisi angka." }; } // 2. Cek apakah bank terdaftar di dalam rules const bank = bankRules[bankCode.toLowerCase()]; if (!bank) { return { isValid: false, message: "Kode bank tidak dikenali atau belum didukung." }; } // 3. Validasi panjang digit nomor rekening const length = cleanAccountNo.length; if (length < bank.minLength || length > bank.maxLength) { let digitInfo = bank.minLength === bank.maxLength ? `harus ${bank.minLength}` : `antara ${bank.minLength} sampai ${bank.maxLength}`; return { isValid: false, message: `Nomor rekening ${bank.name} tidak valid. Panjang digit ${digitInfo} angka.` }; } // Jika lolos semua pengecekan return { isValid: true, message: `Nomor rekening ${bank.name} valid.` }; } // ========================================== // CONTOH PENGGUNAAN (TESTING): // ========================================== // Test BCA (Harus 10 digit) console.log(validateBankAccount("bca", "1234567890")); // Output: { isValid: true, message: 'Nomor rekening BCA valid.' } console.log(validateBankAccount("bca", "12345")); // Output: { isValid: false, message: 'Nomor rekening BCA tidak valid. Panjang digit harus 10 angka.' } // Test Mandiri (Harus 13 digit) console.log(validateBankAccount("mandiri", "1230009876543")); // Output: { isValid: true, message: 'Nomor rekening Mandiri valid.' } // Test BRI (Harus 15 digit) console.log(validateBankAccount("bri", "034101000123501")); // Output: { isValid: true, message: 'Nomor rekening BRI valid.' }
